import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error
from sklearn.ensemble import RandomForestRegressor



url = "https://raw.githubusercontent.com/unnatibisht/Uber-Fare-Prediction/main/uber.csv"
df = pd.read_csv(url)
df.head()


df.info()


print(df['pickup_datetime'].dtype)


df['pickup_datetime'] = pd.to_datetime(
    df['pickup_datetime'],
    errors='coerce',
    utc=True
)


print(df['pickup_datetime'].dtype)


df.describe()


df.isnull().sum()


df.dropna(inplace=True)


df.isnull().sum()



q_low = df['fare_amount'].quantile(0.01)
q_hi = df['fare_amount'].quantile(0.99)
print(q_low, q_hi)


df


plt.boxplot(df['fare_amount'])
plt.title("Fare Amount Before Removing Outliers")
plt.ylabel("Fare Amount")
plt.show()



df = df[
    (df['fare_amount'] < q_hi) &
    (df['fare_amount'] > q_low)
]



df.isnull().sum()


plt.boxplot(df['fare_amount'])
plt.title("Fare Amount After Removing Outliers")
plt.ylabel("Fare Amount")
plt.show()


df.shape


df = df.drop(
    columns=['key', 'Unnamed: 0'],
    errors='ignore'
)
df.head()



df['hour'] = df['pickup_datetime'].dt.hour
df['day'] = df['pickup_datetime'].dt.day
df['month'] = df['pickup_datetime'].dt.month
df.head()


df.drop('pickup_datetime', axis=1, inplace=True)


# Take X as predictor variable
x = df.drop('fare_amount', axis=1)
# Take Y as target variable
y = df['fare_amount']


print(x.shape)
print(y.shape)


x.head()


print(x.dtypes)


x = x.select_dtypes(include=np.number)
x.shape



x_train, x_test, y_train, y_test = train_test_split(
    x,
    y,
    test_size=0.2,
    random_state=1
)
print(x_train.shape)
print(x_test.shape)



lrmodel = LinearRegression()
lrmodel.fit(x_train, y_train)
# Prediction
predict = lrmodel.predict(x_test)
# Check Error
lrmodelrmse = np.sqrt(
    mean_squared_error(y_test, predict)
)
print("RMSE error for the model is:", lrmodelrmse)



rfrmodel = RandomForestRegressor(
    n_estimators=20,
    max_depth=10,
    random_state=101,
    n_jobs=-1
)
# Fit the Forest
rfrmodel.fit(x_train, y_train)
# Prediction
rfrmodel_pred = rfrmodel.predict(x_test)
# Calculate Error
rfrmodel_rmse = np.sqrt(
    mean_squared_error(y_test, rfrmodel_pred)
)
print("RMSE value for Random Forest is:", rfrmodel_rmse)



print("Linear Regression RMSE:", lrmodelrmse)
print("Random Forest RMSE:", rfrmodel_rmse)
if lrmodelrmse < rfrmodel_rmse:
    print("Linear Regression performs better")
else:
    print("Random Forest performs better")
