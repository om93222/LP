
public class A5_Knapsack {

    public static int knapsack(int capacity, int[] weight,
            int[] value, int n) {

        int[][] dp = new int[n + 1][capacity + 1];

        for (int i = 0; i <= n; i++) {

            for (int w = 0; w <= capacity; w++) {

                if (i == 0 || w == 0) {
                    dp[i][w] = 0;
                }

                else if (weight[i - 1] <= w) {

                    dp[i][w] = Math.max(
                            value[i - 1] +
                                    dp[i - 1][w - weight[i - 1]],

                            dp[i - 1][w]);
                }

                else {
                    dp[i][w] = dp[i - 1][w];
                }
            }
        }

        return dp[n][capacity];
    }

    public static void main(String[] args) {

        int[] weight = { 10, 20, 30 };
        int[] value = { 60, 100, 120 };

        int capacity = 50;

        int n = value.length;

        int maxValue = knapsack(capacity, weight, value, n);

        System.out.println("Maximum value obtained = " + maxValue);
    }
}
