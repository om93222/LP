
public class Fibonacci {

    // Recursive: F(n) = F(n-1) + F(n-2)
    public static long fibRecursive(int n) {
        if (n <= 1) {
            return n;
        }
        return fibRecursive(n - 1) + fibRecursive(n - 2);
    }

    // Non-recursive (iterative): uses a loop
    public static long fibIterative(int n) {
        if (n <= 1) {
            return n;
        }
        long previous = 0;
        long current = 1;
        for (int i = 2; i <= n; i++) {
            long next = previous + current;
            previous = current;
            current = next;
        }
        return current;
    }

    public static void main(String[] args) {
        int n = 5;  // change this value

        System.out.println("Recursive:     Fibonacci(" + n + ") = " + fibRecursive(n));
        System.out.println("Non-recursive: Fibonacci(" + n + ") = " + fibIterative(n));

        System.out.print("Series: ");
        for (int i = 0; i < n; i++) {
            System.out.print(fibIterative(i) + " ");
        }
        System.out.println();
    }
}
